        function calculateCMV() {
            const item = document.getElementById('cmv-item').value;
            const cost = parseFloat(document.getElementById('cmv-cost').value);
            const target = parseFloat(document.getElementById('cmv-target').value);
            if(!item || isNaN(cost) || isNaN(target)) return;
            const price = cost / (target / 100);
            const margin = ((price - cost) / price) * 100;
            document.getElementById('res-price').innerText = `R$ ${price.toFixed(2)}`;
            document.getElementById('res-margin').innerText = `${margin.toFixed(1)}%`;
            document.getElementById('cmv-result').classList.remove('hidden');
        }

        let employees = [];
        function addEmployee() {
            const name = document.getElementById('emp-name').value;
            const faltas = parseInt(document.getElementById('emp-faltas').value);
            if(!name || isNaN(faltas)) return;
            employees.push({name, faltas});

        }
